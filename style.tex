\usepackage[utf8]{inputenc}
\usepackage[T1]{fontenc}
\usepackage{lmodern}
\usepackage[italian,english]{babel}   % English is the main document language;
                                       % Italian is only used for the cover page
\usepackage[a4paper,margin=2.5cm]{geometry}
\usepackage{pdflscape}   % landscape pages for full-page figures (e.g. the Qsys block diagram)

\usepackage{indentfirst}   % also indent the first paragraph after a heading
\setlength{\parindent}{1em}
\setlength{\parskip}{0.6em}

% Lets TeX loosen a line as a last resort, instead of producing an overfull
% \hbox, when an unbreakable inline |code| token (see \lstMakeShortInline
% below) doesn't quite fit at the end of it. Purely a safety valve: it only
% engages when the normal stretch budget is exhausted, so ordinary paragraphs
% are unaffected.
\emergencystretch=1in

\usepackage{graphicx}
\graphicspath{{img/}}          % \includegraphics{filename} looks inside img/
\usepackage{adjustbox}         % lets text auto-scale to fit a given width (e.g. thesis title)

% SVG figures: \includesvg{name} converts img/name.svg to PDF (and PDF_TEX) at
% compile time via inkscape, so editing an SVG only needs a recompile. The
% generated files land in the build directory (see .latexmkrc).
\usepackage{svg}
\svgsetup{
    inkscapeexe=inkscape,
    inkscapearea=drawing,
    inkscapeversion=1,
    inkscapelatex=false,   % render SVG text with inkscape's own fonts, so
                           % underscores etc. in the SVG need no LaTeX escaping
    inkscapepath=build/svg-inkscape/,   % generated PDFs land in the build directory
}
\svgpath{img/}

\usepackage{booktabs}          % \toprule/\midrule/\bottomrule for clean tables
\usepackage{colortbl}          % \arrayrulecolor, for \lightmidrule below
\newcommand{\lightmidrule}{\arrayrulecolor{black!40}\midrule\arrayrulecolor{black}}
% Usage: \lightmidrule -- a lighter separator between the rows of a table,
% for tables whose cells wrap onto multiple lines, where a full-weight
% \midrule on every row looks too much like a spreadsheet grid. Leaves
% \toprule/\bottomrule (the table's own outer frame) solid black.
\usepackage{tabularx}          % X columns auto-wrap long content (e.g. file paths)
\usepackage{seqsplit}          % breaks long unbroken strings (paths, tokens) that don't wrap on their own
\usepackage{float}             % [H] placement: keep floats exactly where they are defined

\usepackage{xcolor}
\newcommand{\todo}[1]{\textbf{\textcolor{red}{[TODO: #1]}}}   % visible marker for unresolved text
\newcommand{\term}[1]{\textsf{#1}}   % hardware/product/protocol names and acronyms (FPGA, DE1-SoC, RGB565, ...)
\newcommand{\sysname}[1]{\textsf{\itshape #1}}   % this thesis's own named systems (FPGAlix, FPGAsteel)

% Glossary (Appendix C): the key (term) and the body (definition) of each entry
% are styled centrally here, so a global look change needs editing only this file.
\newcommand{\glossaryKey}[1]{\textsf{\textbf{#1}}}   % glossary entry key (the term)
\newcommand{\glossaryDescription}[1]{\normalfont #1} % glossary entry body (the definition)

\usepackage[normalem]{ulem}
\newcommand{\reread}[1]{\sout{#1}}   % marks text to go back and review
\newcommand{\unresolved}[1]{\textcolor{red}{#1}}   % marks citations NOT in the global bibliography (Appendix B); decide what to do with them on reread

\newcommand{\longsection}[2]{%
    \section[#1]{#1}%
    \vspace{-2.6ex}%
    {\large\itshape #2\par}%
    \vspace{1ex}%
}
% Usage: \longsection{Short title}{Detail shown as a subtitle line below it}
% For section titles too long for the running header (e.g. "Title (code_name)"
% or "Title: detail") -- the short title alone goes in the header/TOC, the
% detail is typeset as a subtitle right under the heading in the body.

\usepackage{listings}

\definecolor{friendcomment}{HTML}{61A1B0}   % comment (blue-gray, italic)
\definecolor{friendrule}{HTML}{D0D7DE}      % frame border

\lstdefinestyle{githubcpp}{
    language=C++,
    basicstyle=\ttfamily\small\color{black},
    commentstyle=\color{friendcomment}\itshape,
    numberstyle=\tiny\color{gray},
    identifierstyle=\color{black},
    breaklines=true,
    breakatwhitespace=false,
    columns=fullflexible,
    keepspaces=true,
    showstringspaces=false,
    tabsize=4,
    frame=single,
    rulecolor=\color{friendrule},
    framesep=5pt,
    xleftmargin=4pt,
    xrightmargin=4pt,
    captionpos=b,
    numbers=left,
    numbersep=8pt,
}
\lstset{style=githubcpp}

% (e.g. export PATH="..."), where breaking only at whitespace causes overflow:

    basicstyle=\ttfamily\scriptsize\color{black},
    commentstyle=\color{friendcomment}\itshape,
    identifierstyle=\color{black},
    breaklines=true,
    breakatwhitespace=false,

    showstringspaces=false,
    tabsize=4,
    frame=single,

    xrightmargin=4pt,
    captionpos=b,
    numbers=none,
}
\lstnewenvironment{shellcodepath}{\lstset{style=githubbashpath}}{}

%   \begin{shellcode}            <- normal commands / word lists (safe wrapping)

    showstringspaces=false,
    breaklines=true,
    breakatwhitespace=false,
}
\lstMakeShortInline[style=githubcppinline]|
